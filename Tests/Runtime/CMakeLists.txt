cmake_minimum_required(VERSION 3.20)
project(PnamRuntimeTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
enable_testing()
get_filename_component(PLUGIN_ROOT ${CMAKE_CURRENT_LIST_DIR}/../.. ABSOLUTE)
set(CORE ${PLUGIN_ROOT}/external/nam_core_parametric)
set(AUDIO ${PLUGIN_ROOT}/external/AudioDSPTools)
file(GLOB CORE_SOURCES CONFIGURE_DEPENDS ${CORE}/NAM/*.cpp ${CORE}/NAM/*/*.cpp)
add_library(pnam_core OBJECT ${CORE_SOURCES})
target_compile_definitions(pnam_core PUBLIC NAM_SAMPLE_FLOAT=1 NAM_ENABLE_A2_FAST=1
    EIGEN_MPL2_ONLY=1 EIGEN_NO_DEBUG=1)
if(WIN32)
    target_compile_definitions(pnam_core PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN)
endif()
target_include_directories(pnam_core SYSTEM PUBLIC ${CORE} ${CORE}/NAM
    ${CORE}/Dependencies/eigen ${CORE}/Dependencies/nlohmann ${AUDIO})
add_executable(pnam_fast_path_test ../PnamFastPathTest.cpp)
target_link_libraries(pnam_fast_path_test PRIVATE pnam_core)
foreach(pair IN ITEMS "hyper_a2_engine_test|HyperA2EngineTest" "pnam_benchmark|PnamBenchmark")
    string(REPLACE "|" ";" parts ${pair})
    list(GET parts 0 target)
    list(GET parts 1 source)
    add_executable(${target} ../${source}.cpp ${PLUGIN_ROOT}/Source/DSP/NamModelEngine.cpp)
    target_include_directories(${target} PRIVATE ${PLUGIN_ROOT}/Source)
    target_link_libraries(${target} PRIVATE pnam_core)
endforeach()
set(PNAM_FIXTURES "" CACHE PATH "Synthetic PNAM fixture directory")
if(EXISTS ${PNAM_FIXTURES}/manifest.json)
    add_test(NAME fast_automation COMMAND pnam_fast_path_test ${PNAM_FIXTURES})
    add_test(NAME engine_parity COMMAND hyper_a2_engine_test ${PNAM_FIXTURES})
endif()
